## 範例

```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import matplotlib.font_manager as fm
import random
from datetime import date, timedelta

# ── 步驟一：產生模擬資料 ──────────────────────────────────────────
CATEGORIES = {
    "餐飲": (50, 300),
    "交通": (20, 200),
    "娛樂": (100, 800),
    "購物": (200, 2000),
    "醫療": (150, 1500),
    "帳單": (500, 3000),
}

def generate_expenses(n=200, seed=42):
    random.seed(seed)
    np.random.seed(seed)
    today = date.today()
    start = today - timedelta(days=180)
    records = []
    for _ in range(n):
        day = start + timedelta(days=random.randint(0, 180))
        cat = random.choice(list(CATEGORIES.keys()))
        lo, hi = CATEGORIES[cat]
        amount = round(random.uniform(lo, hi), 0)
        records.append({
            "date": day.strftime("%Y-%m-%d"),
            "category": cat,
            "amount": int(amount),
            "note": "",
        })
    return pd.DataFrame(records)

df = generate_expenses()
df.to_csv("expenses.csv", index=False, encoding="utf-8-sig")
print(f"已產生 {len(df)} 筆資料，存入 expenses.csv")
```

## 提示
- `df["date"].dt.to_period("M").astype(str)` 可提取 `YYYY-MM` 格式月份
- `groupby(...).sum().unstack(fill_value=0)` 可建立交叉表格（月份 × 類別）
- 圓餅圖：`ax.pie(sizes, labels=labels, autopct="%1.1f%%", startangle=90)`
- 堆疊長條圖：對每個類別逐一呼叫 `ax.bar(x, height, bottom=cumulative)`，或使用 `DataFrame.plot(kind="bar", stacked=True, ax=ax)`
- 中文字型問題：在 Linux 可用 `plt.rcParams["font.family"] = "DejaVu Sans"` 並將中文標籤改為英文，或安裝 `noto-fonts-cjk` 套件
- `f"${value:,.0f}"` 可格式化金額字串（加千分位逗號）
