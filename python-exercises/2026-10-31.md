```python
feed = feedparser.parse(url)
for entry in feed.entries:
    print(entry.title)        # 標題
    print(entry.summary)      # 摘要（可能不存在）
    print(entry.published)    # 發布時間（字串，格式不一）
    print(entry.link)         # 連結
```

### 2. 用 TextBlob 計算情緒

```python
from textblob import TextBlob

blob = TextBlob("Apple stock surges to record high on strong earnings")
polarity   = blob.sentiment.polarity     # -1.0（極負）到 +1.0（極正）
subjectivity = blob.sentiment.subjectivity # 0.0（客觀）到 1.0（主觀）
```

依據 polarity 分類：
- `polarity > 0.05` → **正面（Positive）**
- `polarity < -0.05` → **負面（Negative）**
- 其他 → **中性（Neutral）**

## 範例

```python
import feedparser
from textblob import TextBlob
import pandas as pd
import matplotlib.pyplot as plt
import argparse
from datetime import datetime

RSS_SOURCES = {
    "BBC Top Stories":  "http://feeds.bbci.co.uk/news/rss.xml",
    "BBC Technology":   "http://feeds.bbci.co.uk/news/technology/rss.xml",
    "Reuters Business": "https://feeds.reuters.com/reuters/businessNews",
    "Reuters Science":  "https://feeds.reuters.com/reuters/scienceNews",
}

def classify(polarity: float) -> str:
    if polarity > 0.05:
        return "Positive"
    elif polarity < -0.05:
        return "Negative"
    return "Neutral"

def fetch_all(sources: dict) -> pd.DataFrame:
    rows = []
    for source_name, url in sources.items():
        feed = feedparser.parse(url)
        for entry in feed.entries:
            title = entry.get("title", "")
            blob  = TextBlob(title)
            pol   = blob.sentiment.polarity
            sub   = blob.sentiment.subjectivity
            rows.append({
                "source":      source_name,
                "title":       title,
                "polarity":    pol,
                "subjectivity":sub,
                "sentiment":   classify(pol),
                "link":        entry.get("link", ""),
                "published":   entry.get("published", ""),
            })
    return pd.DataFrame(rows)

def cmd_analyze(args):
    sources = RSS_SOURCES
    if args.source:
        sources = {args.source: RSS_SOURCES[args.source]}

    df = fetch_all(sources)
    now = datetime.utcnow().strftime("%Y-%m-%d %H:%M UTC")
    print(f"\n📰 RSS 新聞情緒分析（{now}）\n")

    # 來源彙整表
    summary = df.groupby("source").apply(lambda g: pd.Series({
        "count":   len(g),
        "pos_pct": (g["sentiment"] == "Positive").mean() * 100,
        "neu_pct": (g["sentiment"] == "Neutral").mean() * 100,
        "neg_pct": (g["sentiment"] == "Negative").mean() * 100,
        "avg_pol": g["polarity"].mean(),
    })).reset_index()

    for _, row in summary.iterrows():
        print(f"{row['source']:<25} {int(row['count']):>4} 篇  "
              f"{row['pos_pct']:>5.1f}%  {row['neu_pct']:>5.1f}%  "
              f"{row['neg_pct']:>5.1f}%  {row['avg_pol']:>+6.2f}")

    print(f"\n📊 整體統計（共 {len(df)} 篇）")
    for sent, label in [("Positive","正面"), ("Neutral","中性"), ("Negative","負面")]:
        n = (df["sentiment"] == sent).sum()
        print(f"  {label}：{n} 篇（{n/len(df)*100:.1f}%）")
    print(f"  整體平均極性：{df['polarity'].mean():+.2f}")

    best  = df.loc[df["polarity"].idxmax()]
    worst = df.loc[df["polarity"].idxmin()]
    print(f"  最高情緒：{best['polarity']:+.2f} → {best['title'][:60]}")
    print(f"  最低情緒：{worst['polarity']:+.2f} → {worst['title'][:60]}")

    if args.export:
        df.to_csv(args.export, index=False, encoding="utf-8-sig")
        print(f"\n✅ 已匯出：{args.export}")

    return df

def cmd_top(args):
    df = fetch_all(RSS_SOURCES)
    if args.sentiment == "positive":
        top = df.nlargest(args.n, "polarity")
        label = "正面"
    else:
        top = df.nsmallest(args.n, "polarity")
        label = "負面"
    print(f"\n🔝 情緒最{label}的 {args.n} 篇新聞標題\n")
    for i, row in top.iterrows():
        print(f"  [{row['polarity']:+.2f}] {row['title']}")
        print(f"        來源：{row['source']}")
```

## 提示

- **TextBlob 初次使用須下載語料庫**：
  ```bash
  python -m textblob.download_corpora
  ```
- **feedparser 無需 requests**：直接傳 URL，內部自行處理 HTTP；若需要自訂 User-Agent：
  ```python
  feedparser.USER_AGENT = "MySentimentBot/1.0"
  ```
- **RSS 來源可能暫時不可用**：加 `try/except` 跳過失敗來源：
  ```python
  try:
      feed = feedparser.parse(url)
      if feed.bozo:  # bozo=True 代表 XML 解析有問題
          raise ValueError("Feed 解析失敗")
  except Exception as e:
      print(f"⚠️  跳過 {source_name}：{e}")
      continue
  ```
- **堆疊長條圖**：
  ```python
  bottom_pos = [0] * len(labels)
  ax.bar(labels, pos_counts, color="green",  label="正面", bottom=bottom_pos)
  ax.bar(labels, neu_counts, color="gray",   label="中性", bottom=pos_counts)
  ax.bar(labels, neg_counts, color="red",    label="負面", bottom=[p+n for p,n in zip(pos_counts, neu_counts)])
  ```
- **散點圖標記形狀**：

## 學習重點

### 與爬蟲／ML 的關聯

| 技術 | 應用場景 |
|------|---------|
| RSS / `feedparser` | 結構化網頁資料來源，比 BeautifulSoup 更穩定（官方格式）|
| `TextBlob.sentiment` | NLP 入門：詞典法（Lexicon-based）情緒分析的代表實作 |
| polarity / subjectivity | ML 特徵工程：可直接作為文字分類模型的輸入特徵 |
| 情緒分類閾值 | 超參數調整：±0.05 的邊界會影響三類別的分佈，可實驗調整 |
| 堆疊長條圖 | EDA 標準圖型：比較多組的組成比例（類別分佈視覺化）|
| 散點圖 polarity vs subjectivity | 雙變數探索：發現「主觀但情緒中性」（如技術報導）的文章群 |
