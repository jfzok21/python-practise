## 具體任務

### 1. 取得城市座標（Geocoding API）

Open-Meteo 提供免費地理編碼 API，透過城市名稱取得經緯度：

```
https://geocoding-api.open-meteo.com/v1/search?name={城市名稱}&count=1&language=en&format=json
```

回傳 JSON，重要欄位：

```json
{
  "results": [
    {
      "name": "Taipei",
      "latitude": 25.04776,
      "longitude": 121.53185,
      "country": "Taiwan",
      "timezone": "Asia/Taipei"
    }
  ]
}
```

### 2. 取得 30 天歷史每日天氣資料

取得座標後，呼叫歷史天氣 API：

```
https://archive-api.open-meteo.com/v1/archive
  ?latitude={lat}
  &longitude={lon}
  &start_date={YYYY-MM-DD}
  &end_date={YYYY-MM-DD}
  &daily=temperature_2m_max,temperature_2m_min,temperature_2m_mean,precipitation_sum,wind_speed_10m_max
  &timezone=auto
```

回傳 JSON，結構如下：

```json
{
  "daily": {
    "time": ["2026-10-03", "2026-10-04", ...],
    "temperature_2m_max": [32.1, 31.5, ...],
    "temperature_2m_min": [22.3, 21.8, ...],
    "temperature_2m_mean": [27.2, 26.6, ...],
    "precipitation_sum": [0.0, 12.4, ...],
    "wind_speed_10m_max": [15.2, 18.7, ...]
  }
}
```

### 3. 實作 CLI 指令


執行 `analyze` 指令後，儲存為 `climate_{城市名稱}.png`，包含三個子圖（垂直排列）：

**上圖：氣溫折線圖**

**中圖：日降雨量長條圖**

**下圖：最大風速折線圖**
- X 軸：日期
- Y 軸：風速（km/h）
- 填滿曲線下方區域（半透明藍）
- 以水平虛線標示強風閾值（50 km/h）

## 範例

```python
import requests
import pandas as pd
import matplotlib.pyplot as plt
import argparse
from datetime import date, timedelta

GEO_URL     = "https://geocoding-api.open-meteo.com/v1/search"
WEATHER_URL = "https://archive-api.open-meteo.com/v1/archive"

def geocode(city: str) -> dict:
    """回傳 {'name', 'country', 'latitude', 'longitude', 'timezone'}"""
    resp = requests.get(GEO_URL, params={"name": city, "count": 1,
                                          "language": "en", "format": "json"},
                        timeout=10)
    resp.raise_for_status()
    results = resp.json().get("results")
    if not results:
        raise ValueError(f"找不到城市：{city}")
    r = results[0]
    return {"name": r["name"], "country": r["country"],
            "latitude": r["latitude"], "longitude": r["longitude"],
            "timezone": r["timezone"]}

def fetch_weather(lat: float, lon: float, days: int = 30,
                  timezone: str = "auto") -> pd.DataFrame:
    """取得近 days 天的每日氣象資料，回傳 DataFrame"""
    end   = date.today() - timedelta(days=1)   # 昨天（最新可用）
    start = end - timedelta(days=days - 1)
    params = {
        "latitude":  lat,
        "longitude": lon,
        "start_date": start.isoformat(),
        "end_date":   end.isoformat(),
        "daily": "temperature_2m_max,temperature_2m_min,temperature_2m_mean,"
                 "precipitation_sum,wind_speed_10m_max",
        "timezone": timezone,
    }
    resp = requests.get(WEATHER_URL, params=params, timeout=15)
    resp.raise_for_status()
    daily = resp.json()["daily"]
    df = pd.DataFrame({
        "date":       pd.to_datetime(daily["time"]),
        "temp_max":   daily["temperature_2m_max"],
        "temp_min":   daily["temperature_2m_min"],
        "temp_mean":  daily["temperature_2m_mean"],
        "precip":     daily["precipitation_sum"],
        "wind_max":   daily["wind_speed_10m_max"],
    })
    df["rolling7"] = df["temp_mean"].rolling(7, min_periods=1).mean()
    return df

def print_summary(df: pd.DataFrame, city_info: dict, days: int):
    start_str = df["date"].iloc[0].strftime("%Y-%m-%d")
    end_str   = df["date"].iloc[-1].strftime("%Y-%m-%d")
    print(f"\n🌤  城市氣候分析：{city_info['name']}, {city_info['country']}"
          f"（{start_str} ~ {end_str}）\n")

    print("📊 氣溫統計（°C）")
    print(f"  {days} 天平均氣溫：{df['temp_mean'].mean():.1f}°C")
    idx_max = df["temp_max"].idxmax()
    idx_min = df["temp_min"].idxmin()
    print(f"  最高氣溫：{df.loc[idx_max, 'temp_max']:.1f}°C"
          f"（{df.loc[idx_max, 'date'].strftime('%Y-%m-%d')}）")
    print(f"  最低氣溫：{df.loc[idx_min, 'temp_min']:.1f}°C"
          f"（{df.loc[idx_min, 'date'].strftime('%Y-%m-%d')}）")

    rain_days = (df["precip"] > 0).sum()
    idx_rain  = df["precip"].idxmax()
    print(f"\n🌧  降雨統計")
    print(f"  總降雨量：{df['precip'].sum():.1f} mm")
    print(f"  降雨天數：{rain_days} 天（{rain_days/len(df)*100:.1f}%）")
    print(f"  最大單日降雨：{df.loc[idx_rain, 'precip']:.1f} mm"
          f"（{df.loc[idx_rain, 'date'].strftime('%Y-%m-%d')}）")

    print(f"\n💨 風速統計")
    print(f"  平均最大風速：{df['wind_max'].mean():.1f} km/h")
    print(f"  最強風速：{df['wind_max'].max():.1f} km/h"
          f"（{df.loc[df['wind_max'].idxmax(), 'date'].strftime('%Y-%m-%d')}）")

def plot_climate(df: pd.DataFrame, city_name: str):
    fig, axes = plt.subplots(3, 1, figsize=(12, 10), sharex=True)
    fig.suptitle(f"氣候趨勢分析：{city_name}（近 {len(df)} 天）", fontsize=14)

    # 上圖：氣溫
    ax = axes[0]
    ax.plot(df["date"], df["temp_max"],  color="#e07b39", label="最高氣溫", lw=1.5)
    ax.plot(df["date"], df["temp_mean"], color="#3b7dd8", label="平均氣溫", lw=2)
    ax.plot(df["date"], df["temp_min"],  color="#56b4e9", label="最低氣溫", lw=1.5)
    ax.plot(df["date"], df["rolling7"],  color="black",   label="7日均溫",
            lw=1.5, linestyle="--", alpha=0.7)
    ax.fill_between(df["date"], df["temp_min"], df["temp_max"], alpha=0.08, color="#3b7dd8")
    ax.set_ylabel("氣溫 (°C)")
    ax.legend(loc="upper right", fontsize=9)
    ax.grid(axis="y", alpha=0.3)

    # 中圖：降雨量
    ax2 = axes[1]
    colors = ["#1565c0" if v > 0 else "#e0e0e0" for v in df["precip"]]
    ax2.bar(df["date"], df["precip"], color=colors, label="日降雨量")
    ax2_r = ax2.twinx()
    ax2_r.plot(df["date"], df["precip"].cumsum(), color="#90caf9",
               lw=1.5, label="累積降雨")
    ax2_r.set_ylabel("累積降雨 (mm)", color="#90caf9")
    ax2.set_ylabel("日降雨量 (mm)")
    ax2.grid(axis="y", alpha=0.3)

    # 下圖：風速
    ax3 = axes[2]
    ax3.plot(df["date"], df["wind_max"], color="#5c6bc0", lw=2, label="最大風速")
    ax3.fill_between(df["date"], 0, df["wind_max"], alpha=0.2, color="#5c6bc0")
    ax3.axhline(50, color="red", linestyle="--", lw=1, alpha=0.7, label="強風警戒 50 km/h")
    ax3.set_ylabel("最大風速 (km/h)")
    ax3.set_xlabel("日期")
    ax3.legend(loc="upper right", fontsize=9)
    ax3.grid(axis="y", alpha=0.3)

    plt.xticks(rotation=30, ha="right")
    plt.tight_layout()
    fname = f"climate_{city_name.replace(' ', '_')}.png"
    plt.savefig(fname, dpi=150, bbox_inches="tight")
    print(f"\n✅ 圖表已儲存：{fname}")
    plt.close()
```

## 提示

- **`date.today() - timedelta(days=1)` 取昨天**：Open-Meteo 歷史資料通常只到前一天，不含今日
- **`rolling(7, min_periods=1)`**：前 6 天資料不足時，用現有資料計算，避免 NaN 開頭
- **雙 Y 軸**：`ax.twinx()` 建立共享 X 軸的第二 Y 軸，適合同圖顯示不同量綱的資料：
  ```python
  ax_right = ax.twinx()
  ax_right.plot(...)
  ax_right.set_ylabel("...")
  ```
- **處理 `None` 值**：API 偶爾回傳 `None`（資料缺失），用 `pd.to_numeric(..., errors='coerce')` 轉為 NaN 再處理：
  ```python
  df["precip"] = pd.to_numeric(df["precip"], errors="coerce").fillna(0)
  ```
- **`compare` 子指令填色**：
  ```python
  ax.fill_between(dates, temp1, temp2,
                  where=(temp1 >= temp2), alpha=0.15, color="red",   label="城市1較熱")
  ax.fill_between(dates, temp1, temp2,
                  where=(temp1 <  temp2), alpha=0.15, color="blue",  label="城市2較熱")
  ```
- **日期軸格式**：
  ```python
  import matplotlib.dates as mdates
  ax.xaxis.set_major_formatter(mdates.DateFormatter("%m/%d"))
  ax.xaxis.set_major_locator(mdates.WeekdayLocator(byweekday=mdates.MO))
  ```

## 學習重點

### 與爬蟲／ML 的關聯

| 技術 | 應用場景 |
|------|---------|
| Open-Meteo REST API | 免費天氣資料來源，可用於環境感測、農業預測等 ML 場景 |
| `rolling(N).mean()` | 時序平滑化：消除噪音、凸顯趨勢，是時序 ML 特徵工程的核心操作 |
| `twinx()` 雙 Y 軸 | EDA 常見技巧：不同量綱資料疊合比較（如降雨量 vs 氣溫）|
| `fill_between` | 強調兩變量的差異，常用於信賴區間、預測誤差範圍視覺化 |
| Geocoding API | 地址正規化（Address Normalization）：將地名轉為結構化座標 |
| `cumsum()` 累積和 | 時序累積特徵，在銷售分析、用電量預測中廣泛使用 |
